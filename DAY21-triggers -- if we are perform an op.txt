triggers :-- if we are perform an operation in table then it will effect in another table
		>>>automatically executed block of command

delimiter $$
create trigger trigger_name
{before|after} {insert|update|delete} on table_name
for each row
begin
-- commands;
end$$
delimiter ;


new   :-- will enter new data into table 
old   :-- will used access the existing data in table 
 

remove the trigger:
---------------------------
>>drop trigger trigger_name;

use ecosystem;

create table bank(
acc bigint,
name varchar(100),
balence int
);

select * from bank;

create table report(
dot datetime default now(),
result varchar(100)
);

select * from bank;

insert into bank values(1,'vijay',100000);

select * from report;

delimiter $$
create trigger t1
after insert on bank
for each row
begin
insert into report(result) values('new account created ...');
end$$
delimiter ;

select * from bank;
