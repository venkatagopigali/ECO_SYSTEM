joins :--

inner join
left join
right join
outer join  :-- will return all values from both tables 
		>>>left join + union + right join
>>>select * from table1 left join table2 on table1.column_name=table2.column_name
union
select * from table1 right join table2 on table1.column_name=table2.column_name;

cross join  :-- will assign each and every row in the first table with each and every in 			second table  i.e. Cartesian product
		>>>select * from table1 cross join table2;


self join    :--   used to mainly perform on single table   

with condition :-- by default return inner join
>>>select * from table1 join table2 on table1.col=table2.col;

without condition :-- by default return cross join
>>>select * from table1 join table2;

natural join    :-- by default inner join
		>>>will remove duplicate columns automatically

>>>select * from table1 natural join table2;


extra operators :-- will used to perform operation between two operands 
==================================================================================
like         :-- will return the result based on pattern string
		>>>select * from table_name where column_name like 'pattern_string';
		wild card functions :--
		-----------------------------------
		% :-- will read zero or more characters at time 
		_ :-- will read only one character
as 
between
union
union all
intersect
except
