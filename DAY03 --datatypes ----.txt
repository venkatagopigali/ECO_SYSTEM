--------------
at last :--
---------
>>>alter table table_name add column column_name datatype;

at middle :--
-----------
>>>alter table table_name add column column_name datatype after exsting_column;

at first :--
-----------------
>>>alter table table_name add column column_name datatype first;

rename column :--
----------------------------
>>>alter table table_name rename column old_column_name to new_column_name;

drop column :--
----------------------------
>>>alter table table_name drop column column_name;

change datatype :--
-----------------------
>>>alter table table_name modify column column_name new_datatype;


rename      :-- will used to change table name 
---------------------------------------------------------------------------
>>>rename table old_table_name to new_table_name;

truncate    :-- will remove total data from table except table  architecture
-------------------------------------------------------------------
>>>truncate table table_name;

--------------------------------------------------------------------
>>>drop table table_name;
