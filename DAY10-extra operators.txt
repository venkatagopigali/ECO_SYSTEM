extra operators :

like
as          :-- will used to change table or column_name temporarily
between     :-- will return all value in between two range of values
		>>>select * from table_name where column_name between start and end_range;
union       :-- will return all values from both table but return only once which are duplicated
		>>>select * from table1
		   union
		   select * from table2;
union all   :-- will return all values from both table with duplicates
		>>>select * from table1
		   union all 
		   select * from table2;
intersect   :-- will return common values in both tables
		>>>select * from table1
		   intersect
		   select * from table2;
except      :-- will return all values from first table except common values with second table 
		>>>select * from table1
		   except
		   select * from table2;
