window functions :-- will used to perform any kind of data

row_number()     :-- will assign a number to row from 1 to n no.of rows
		>>>select *,row_number() over() from table_name;
rank()           :-- will skip sequence of rank when duplicates are occurred
		>>>select *,rank() over(ordered columns) from table_name;
dense_rank()     :-- will not skip the sequence of rank including duplicates are occurred
		>>>select *,dense_rank() over(ordered column) from table_name;
lead()           :-- will used to move the entire to upwards
		>>>select *,lead(column_name,no.of_pos) over() from table_name;
lag()            :-- will used to move the entire column to downward
		>>>select *,lag(column_name,no.of_pos) over() from table_name;
ntile()          :-- will used to make the no.of groups in a table
		>>>select ,ntile(no.of_groups) over() from table_name;


partition by :-- will separate same type of values
